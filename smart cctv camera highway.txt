  <script>
    // Audio Alert Synthesizer
    let audioCtx = null;
    let audioMuted = true;
    function playBeep(freq = 600, duration = 0.15) {
      if (audioMuted) return;
      try {
        if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + duration);
      } catch (e) {}
    }

    document.getElementById('audio-toggle').addEventListener('click', () => {
      audioMuted = !audioMuted;
      document.getElementById('audio-icon').textContent = audioMuted ? '🔇' : '🔊';
    });

    // Clock
    function updateClock() {
      const now = new Date();
      const timeStr = now.toTimeString().split(' ')[0];
      document.getElementById('clock-display').textContent = timeStr;
      document.getElementById('canvas-time').textContent = timeStr + " UTC";
      document.getElementById('canvas-date').textContent = now.toISOString().split('T')[0];
    }
    setInterval(updateClock, 1000);
    updateClock();

    // Canvas Setup
    const canvas = document.getElementById('highway-canvas');
    const ctx = canvas.getContext('2d');
    let showBoxes = true;
    let weatherMode = 'clear';

    document.getElementById('btn-toggle-boxes').addEventListener('click', () => { showBoxes = !showBoxes; });
    document.getElementById('weather-select').addEventListener('change', (e) => {
      weatherMode = e.target.value;
      document.getElementById('weather-label').textContent = weatherMode.toUpperCase() + ' MODE';
    });

    const lanes = [
      { xTop: 310, xBot: 110 },
      { xTop: 340, xBot: 340 },
      { xTop: 370, xBot: 570 }
    ];
    const vehicleTypes = [
      { type: 'Car', color: '#38bdf8', speed: 85 },
      { type: 'SUV', color: '#a855f7', speed: 90 },
      { type: 'Truck', color: '#f59e0b', speed: 65 },
      { type: 'Sedan', color: '#10b981', speed: 78 }
    ];
    const platesPrefix = ['MH 12', 'DL 01', 'KA 05', 'HR 26', 'UP 16', 'GJ 03'];

    let vehicles = [];
    let vehicleIdCounter = 100;
    let totalMonitored = 1482;
    let violationCount = 18;
    let incidentCount = 1;

    function createVehicle(isSpeeding = false, isAccident = false) {
      vehicleIdCounter++;
      const laneIdx = Math.floor(Math.random() * lanes.length);
      const template = vehicleTypes[Math.floor(Math.random() * vehicleTypes.length)];
      let speed = template.speed + (Math.random() * 14 - 7);
      if (isSpeeding) speed = 125 + Math.random() * 25;
      if (isAccident) speed = 0;

      const plate = `${platesPrefix[Math.floor(Math.random() * platesPrefix.length)]} ${String.fromCharCode(65+Math.floor(Math.random()*26))}${String.fromCharCode(65+Math.floor(Math.random()*26))} ${Math.floor(1000+Math.random()*9000)}`;

      return {
        id: vehicleIdCounter,
        type: template.type,
        color: isSpeeding ? '#ef4444' : template.color,
        speed: Math.round(speed),
        lane: laneIdx,
        progress: isAccident ? 0.65 : 0,
        speedRate: (speed / 100) * 0.0035 + 0.001,
        plate: plate,
        isSpeeding: speed > 100,
        isAccident: isAccident,
        audited: false
      };
    }

    for (let i = 0; i < 5; i++) {
      const v = createVehicle();
      v.progress = Math.random() * 0.85;
      vehicles.push(v);
    }

    document.getElementById('btn-spawn-speed').addEventListener('click', () => { vehicles.push(createVehicle(true, false)); });
    document.getElementById('btn-spawn-accident').addEventListener('click', () => {
      const v = createVehicle(false, true);
      vehicles.push(v);
      addIncidentAlert(v);
    });

    const logContainer = document.getElementById('event-log-container');
    function logViolation(v) {
      violationCount++;
      document.getElementById('stat-violations').textContent = violationCount;
      updateANPR(v, 'OVERSPEED VIOLATION', 'text-red-400');
      playBeep(900, 0.2);

      const item = document.createElement('div');
      item.className = 'p-2.5 rounded-xl bg-red-950/40 border border-red-500/40 flex items-start justify-between animate-pulse';
      item.innerHTML = `<div><div class="flex items-center gap-2"><span class="px-1.5 py-0.5 rounded text-[10px] font-bold bg-red-600 text-white">SPEED LIMIT EXCEEDED</span><span class="font-mono text-gray-200 font-bold">${v.plate}</span></div><p class="text-gray-300 mt-1">Clocked at <strong class="text-red-400">${v.speed} km/h</strong> (Limit: 100 km/h). Auto E-Challan generated.</p></div><span class="text-[10px] text-gray-400 font-mono">${new Date().toTimeString().split(' ')[0]}</span>`;
      logContainer.insertBefore(item, logContainer.firstChild);
    }

    function addIncidentAlert(v) {
      incidentCount++;
      document.getElementById('stat-incidents').textContent = incidentCount;
      playBeep(400, 0.4);

      const item = document.createElement('div');
      item.className = 'p-2.5 rounded-xl bg-amber-950/40 border border-amber-500/50 flex items-start justify-between';
      item.innerHTML = `<div><div class="flex items-center gap-2"><span class="px-1.5 py-0.5 rounded text-[10px] font-bold bg-amber-500 text-black">STALLED VEHICLE / ACCIDENT</span><span class="font-mono text-amber-200">${v.plate}</span></div><p class="text-gray-300 mt-1">Hazard: Zero velocity in Lane ${v.lane + 1}. Patrol unit dispatched.</p></div><span class="text-[10px] text-gray-400 font-mono">${new Date().toTimeString().split(' ')[0]}</span>`;
      logContainer.insertBefore(item, logContainer.firstChild);
    }

    function updateANPR(v, statusText, statusClass) {
      document.getElementById('anpr-plate-display').textContent = v.plate;
      document.getElementById('anpr-class').textContent = `${v.type} (#${v.id})`;
      document.getElementById('anpr-speed').textContent = `${v.speed} km/h`;
      const s = document.getElementById('anpr-status');
      s.textContent = statusText;
      s.className = `${statusClass} font-semibold`;
    }

    let roadOffset = 0;
    function render() {
      ctx.fillStyle = weatherMode === 'night' ? '#05070d' : '#1a202c';
      ctx.fillRect(0, 0, canvas.width, canvas.height);

      const horizonY = 110;
      ctx.fillStyle = weatherMode === 'night' ? '#111827' : '#272e3d';
      ctx.beginPath();
      ctx.moveTo(270, horizonY); ctx.lineTo(410, horizonY);
      ctx.lineTo(670, canvas.height); ctx.lineTo(10, canvas.height);
      ctx.closePath(); ctx.fill();

      // Road Borders
      ctx.strokeStyle = '#e2e8f0'; ctx.lineWidth = 3;
      ctx.beginPath(); ctx.moveTo(270, horizonY); ctx.lineTo(10, canvas.height); ctx.stroke();
      ctx.beginPath(); ctx.moveTo(410, horizonY); ctx.lineTo(670, canvas.height); ctx.stroke();

      // Dashed Dividers
      roadOffset = (roadOffset + 4) % 40;
      ctx.strokeStyle = weatherMode === 'night' ? 'rgba(255,255,255,0.4)' : '#fbbf24';
      ctx.setLineDash([15, 20]);
      ctx.lineDashOffset = -roadOffset;
      ctx.lineWidth = 2;
      ctx.beginPath(); ctx.moveTo(317, horizonY); ctx.lineTo(230, canvas.height); ctx.stroke();
      ctx.beginPath(); ctx.moveTo(363, horizonY); ctx.lineTo(450, canvas.height); ctx.stroke();
      ctx.setLineDash([]);

      // Tripwires
      ctx.strokeStyle = 'rgba(234, 179, 8, 0.4)';
      ctx.beginPath(); ctx.moveTo(180, 240); ctx.lineTo(500, 240); ctx.stroke();
      ctx.strokeStyle = 'rgba(239, 68, 68, 0.5)';
      ctx.beginPath(); ctx.moveTo(90, 360); ctx.lineTo(590, 360); ctx.stroke();

      vehicles.sort((a, b) => a.progress - b.progress);

      for (let i = vehicles.length - 1; i >= 0; i--) {
        const v = vehicles[i];
        if (!v.isAccident) v.progress += v.speedRate;

        const curX = lanes[v.lane].xTop + (lanes[v.lane].xBot - lanes[v.lane].xTop) * v.progress;
        const curY = horizonY + (canvas.height - horizonY) * (v.progress * v.progress);
        const scale = 0.25 + v.progress * 0.95;
        const w = 38 * scale; const h = 58 * scale;

        if (curY >= 355 && !v.audited) {
          v.audited = true;
          totalMonitored++;
          document.getElementById('stat-total-vehicles').textContent = totalMonitored.toLocaleString();
          if (v.isSpeeding) logViolation(v);
          else updateANPR(v, 'COMPLIANT', 'text-emerald-400');
        }

        ctx.fillStyle = v.color;
        ctx.beginPath();
        ctx.roundRect(curX - w/2, curY - h/2, w, h, 4 * scale);
        ctx.fill();

        if (showBoxes) {
          ctx.strokeStyle = v.isSpeeding ? '#ef4444' : (v.isAccident ? '#f59e0b' : '#10b981');
          ctx.lineWidth = 1.5;
          ctx.strokeRect(curX - w/2 - 4, curY - h/2 - 4, w + 8, h + 8);
          ctx.fillStyle = v.isSpeeding ? '#ef4444' : '#10b981';
          ctx.fillRect(curX - w/2 - 4, curY - h/2 - 18, w + 8, 14);
          ctx.fillStyle = '#000';
          ctx.font = 'bold 9px monospace';
          ctx.fillText(`${v.type} ${v.speed}k`, curX - w/2 - 2, curY - h/2 - 7);
        }

        if (v.progress > 1.15) {
          vehicles.splice(i, 1);
          vehicles.push(createVehicle());
        }
      }

      requestAnimationFrame(render);
    }
    render();
  </script>
